        let currentQuestionIndex = 0;
        let score = 0;
        let answeredCurrent = false;
        let firstAttempt = true;

        const buttonColors = [
            { bg: "bg-rose-500", hover: "hover:bg-rose-600", border: "border-rose-700", text: "text-white" },
            { bg: "bg-sky-500", hover: "hover:bg-sky-600", border: "border-sky-700", text: "text-white" },
            { bg: "bg-amber-500", hover: "hover:bg-amber-600", border: "border-amber-700", text: "text-white" },
            { bg: "bg-purple-500", hover: "hover:bg-purple-600", border: "border-purple-700", text: "text-white" }
        ];

        function loadQuestion() {
            answeredCurrent = false;
            firstAttempt = true;

            const q = questions[currentQuestionIndex];
            
            document.getElementById("progress").textContent = `Question ${currentQuestionIndex + 1} of ${questions.length}`;
            document.getElementById("question-text").textContent = q.question;

            // Illustration visibility
            const illusContainer = document.getElementById("illustration-container");
            if (q.showIllustration) {
                illusContainer.classList.remove("hidden");
            } else {
                illusContainer.classList.add("hidden");
            }

            // Options container setup
            const optionsContainer = document.getElementById("options-container");
            optionsContainer.innerHTML = "";
            optionsContainer.classList.remove("hidden");

            q.options.forEach((optText, index) => {
                const color = buttonColors[index % buttonColors.length];
                const btn = document.createElement("button");
                btn.className = `${color.bg} ${color.hover} ${color.text} font-bold py-4 px-4 rounded-2xl text-xl border-b-4 ${color.border} btn-bounce shadow-custom-sm text-center w-full transition-all flex items-center justify-center min-h-[70px]`;
                btn.innerHTML = `<span>${optText}</span>`;
                btn.onclick = () => checkAnswer(index, btn);
                optionsContainer.appendChild(btn);
            });

            // Reset Feedback area
            const feedbackArea = document.getElementById("feedback-area");
            feedbackArea.classList.add("hidden");
            feedbackArea.className = "hidden rounded-2xl p-4 sm:p-5 text-center text-lg sm:text-xl font-bold border-4 transition-all";
        }

        function checkAnswer(selectedIndex, btnElement) {
            if (answeredCurrent) return;

            const q = questions[currentQuestionIndex];
            const feedbackArea = document.getElementById("feedback-area");
            const feedbackTitle = document.getElementById("feedback-title");
            const feedbackMsg = document.getElementById("feedback-message");
            const optionsContainer = document.getElementById("options-container");

            feedbackArea.classList.remove("hidden");

            if (selectedIndex === q.correct) {
                answeredCurrent = true;
                if (firstAttempt) {
                    score++;
                    document.getElementById("score").textContent = score;
                }

                feedbackArea.classList.add("bg-emerald-100", "border-emerald-400", "text-emerald-900");
                feedbackTitle.textContent = "🎉 Correct!";
                feedbackMsg.textContent = q.explanation;
                
                // Highlight correct button
                btnElement.classList.add("ring-4", "ring-emerald-400");

                // Disable non-selected buttons
                Array.from(optionsContainer.children).forEach((btn, idx) => {
                    if (idx !== selectedIndex) {
                        btn.style.opacity = "0.5";
                        btn.disabled = true;
                    }
                });
            } else {
                firstAttempt = false;
                feedbackArea.classList.add("bg-rose-100", "border-rose-400", "text-rose-900");
                feedbackTitle.textContent = "❌ Try again!";
                feedbackMsg.textContent = "That is not quite right. Give it another shot!";
                
                // Visually mark wrong attempt button
                btnElement.style.opacity = "0.4";
                btnElement.disabled = true;
            }
        }

        function nextQuestion() {
            currentQuestionIndex++;
            if (currentQuestionIndex < questions.length) {
                loadQuestion();
            } else {
                showEndScreen();
            }
        }

        function showEndScreen() {
            document.getElementById("question-container").classList.add("hidden");
            document.getElementById("quiz-header").classList.add("hidden");
            
            const endContainer = document.getElementById("end-container");
            endContainer.classList.remove("hidden");

            document.getElementById("final-score").textContent = score;

            const msgElement = document.getElementById("encouragement-msg");
            if (score === 8) {
                msgElement.textContent = "🌟 Wow! Perfect Score! You are an Ellipsoid Expert!";
            } else if (score >= 5) {
                msgElement.textContent = "👍 Great Job! You know your 3D shapes really well!";
            } else {
                msgElement.textContent = "💪 Good Effort! Try again to get a perfect score!";
            }
        }

        function restartQuiz() {
            score = 0;
            currentQuestionIndex = 0;
            document.getElementById("score").textContent = "0";

            document.getElementById("end-container").classList.add("hidden");
            document.getElementById("quiz-header").classList.remove("hidden");
            document.getElementById("question-container").classList.remove("hidden");

            loadQuestion();
        }

        // Initialize quiz on load
        loadQuestion();

I have created an interactive 8-question quiz web app designed specifically for primary school pupils exploring the 3D ellipsoid shape.

### Key Features Included:
- **Child-Friendly Design:** Features bold fonts, cheerful colors, high contrast, and large touch-friendly buttons.
- **Progression & Scoring:** Questions range from simple shape identification to detailed mathematical properties (faces, vertices, edges). It awards 1 point for first-attempt correct answers and displays real-time progress (`Question X of 8`).
- **3D SVG Illustration:** Features an accurate, visually clear SVG illustration of an elongated ellipsoid (rugby ball shape) for Question 1.
- **Instant Feedback:** Displays clear "Correct!" or "Try again!" prompts after each attempt with mathematical explanations.
- **End Summary:** Shows final score out of 8, an encouraging badge message, and a "Play Again" button.
